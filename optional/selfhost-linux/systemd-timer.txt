# systemd timer — more robust than cron (logs to journald, retries, survives reboots)
#
# Install:
#   sudo cp volcano-monitor.service volcano-monitor.timer /etc/systemd/system/
#   sudo systemctl daemon-reload
#   sudo systemctl enable --now volcano-monitor.timer
#
# Check:
#   systemctl list-timers volcano-monitor.timer     # when it next runs
#   journalctl -u volcano-monitor.service -f        # live output
#
# Adjust User= and the paths to match your setup.

# ══════════ /etc/systemd/system/volcano-monitor.service ══════════
[Unit]
Description=Anak Krakatau activity monitor (MAGMA Indonesia)
After=network-online.target
Wants=network-online.target

[Service]
Type=oneshot
User=youruser
WorkingDirectory=/home/youruser/krakatau
ExecStart=/usr/bin/python3 /home/youruser/krakatau/volcano_monitor.py \
          --state /home/youruser/krakatau/state.json \
          --log   /home/youruser/krakatau/history.jsonl
TimeoutStartSec=180
# success = exit 0 (no change) or 2 (alert fired); both are "worked as intended"
SuccessExitStatus=2


# ══════════ /etc/systemd/system/volcano-monitor.timer ══════════
[Unit]
Description=Run the Anak Krakatau monitor every 15 minutes

[Timer]
OnBootSec=2min
OnUnitActiveSec=15min
AccuracySec=1min
RandomizedDelaySec=45
Persistent=true          # run once immediately if a scheduled run was missed

[Install]
WantedBy=timers.target
